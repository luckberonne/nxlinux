cmake_minimum_required(VERSION 3.20)
project(nxlinux C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_C_STANDARD 17)
set(DYNARMIC_FRONTENDS "A64" CACHE STRING "" FORCE)
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_USE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/dynarmic EXCLUDE_FROM_ALL)
add_subdirectory(spikes/jit_hello)
add_subdirectory(spikes/jit_dirty)
enable_testing()
add_executable(test_hle_call tests/test_hle_call.c src/host/hle_call.c)
add_test(NAME hle_call COMMAND test_hle_call)
add_executable(t_tex tests/gfx/t_tex.c src/gfx/tex/nvn_deswizzle.c src/gfx/nvn/nvn_tex.c)
target_link_libraries(t_tex PRIVATE m)
target_compile_options(t_tex PRIVATE -Wall -Wextra)
add_test(NAME tex COMMAND t_tex)
add_library(nxcpu STATIC src/host/cpu.cpp)
target_link_libraries(nxcpu PUBLIC dynarmic)
add_executable(test_cpu_atomic tests/test_cpu_atomic.cpp)
target_link_libraries(test_cpu_atomic PRIVATE nxcpu pthread)
add_test(NAME cpu_atomic COMMAND test_cpu_atomic)
add_library(nxhost STATIC src/host/guest_mem.c src/host/hle_stub.c src/host/module.c src/host/hle_call.c)
target_link_libraries(nxhost PUBLIC nxcpu)
set(MOD_A ${CMAKE_BINARY_DIR}/mod_a.so)
add_custom_command(OUTPUT ${MOD_A}
    COMMAND clang --target=aarch64-linux-gnu -O1 -fPIC -nostdlib -shared -fuse-ld=lld -o ${MOD_A} ${CMAKE_SOURCE_DIR}/tests/guest/mod_a.c
    DEPENDS tests/guest/mod_a.c)
add_custom_target(mod_a_so DEPENDS ${MOD_A})
add_executable(test_module tests/test_module.cpp)
target_link_libraries(test_module PRIVATE nxhost pthread)
add_dependencies(test_module mod_a_so)
add_test(NAME module COMMAND test_module ${MOD_A})
target_sources(nxhost PRIVATE src/host/kernel.c src/host/mem.c)
set(GUEST_THREADS ${CMAKE_BINARY_DIR}/threads.so)
add_custom_command(OUTPUT ${GUEST_THREADS}
    COMMAND clang --target=aarch64-linux-gnu -O1 -fPIC -nostdlib -shared -fuse-ld=lld -o ${GUEST_THREADS} ${CMAKE_SOURCE_DIR}/tests/guest/threads.c
    DEPENDS tests/guest/threads.c)
add_custom_target(threads_so DEPENDS ${GUEST_THREADS})
add_executable(test_kernel tests/test_kernel.cpp)
target_link_libraries(test_kernel PRIVATE nxhost pthread)
add_dependencies(test_kernel threads_so)
add_test(NAME kernel COMMAND test_kernel ${GUEST_THREADS})
set(GUEST_MEMORY ${CMAKE_BINARY_DIR}/memory.so)
add_custom_command(OUTPUT ${GUEST_MEMORY}
    COMMAND clang --target=aarch64-linux-gnu -O1 -fPIC -nostdlib -shared -fuse-ld=lld -o ${GUEST_MEMORY} ${CMAKE_SOURCE_DIR}/tests/guest/memory.c
    DEPENDS tests/guest/memory.c)
add_custom_target(memory_so DEPENDS ${GUEST_MEMORY})
add_executable(test_mem tests/test_mem.cpp)
target_link_libraries(test_mem PRIVATE nxhost pthread)
add_dependencies(test_mem memory_so)
add_test(NAME mem COMMAND test_mem ${GUEST_MEMORY})
add_executable(vkprobe tools/vkprobe.c)
target_include_directories(vkprobe PRIVATE third_party/vulkan-headers/include)
target_link_libraries(vkprobe PRIVATE vulkan)
